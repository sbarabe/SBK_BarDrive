93 bool update(uint32_t syncTime = millis())
94 {
95 _currentTime = syncTime;
96
97 if (_queueEntryTimingPending)
98 {
99 _queueEntryStartedAt = _currentTime;
100 _queueEntryTimingPending = false;
101 }
102
103 if (_isRunning && _currentFunc && _activeBlockEmissionStopAfter > 0 &&
104 _currentTime - _queueEntryStartedAt >= _activeBlockEmissionStopAfter)
105 {
106 if (_currentFunc == &SBK_BarMeterAnimations::_mirrorBlocks ||
107 _currentFunc == &SBK_BarMeterAnimations::_scrollingBlocks)
108 {
109 // Stop producing blocks at the deadline, but keep updating until
110 // every block already in flight has naturally left the display.
111 _emittingBlocksEnabled = false;
112 _loop = false;
113 }
114 _activeBlockEmissionStopAfter = 0;
115 }
116
117 if (_isRunning && _currentFunc && _queueEntryDuration > 0 &&
118 _currentTime - _queueEntryStartedAt >= _queueEntryDuration)
119 {
120 if (_queueActive)
121 _startNextQueuedAnimation();
122 else
123 {
124 _isRunning = false;
125 _isPaused = false;
126 _loop = false;
127 _isLoopingNow = false;
128 _currentFunc = nullptr;
129 _init = true;
130 _queueEntryDuration = 0;
131 }
132 return _isRunning;
133 }
134
135 if (!_isRunning || _isPaused || !_currentFunc)
136 return _queueActive;
137
138 if ((this->*_currentFunc)())
139 {
140 if (_loop)
141 {
142 if (_skipPending)
143 _isLoopingNow = false;
144 else
145 {
146 _isLoopingNow = true; // mark it so user can react
147 _init = true;
148 }
149 }
150 else
151 {
152 if (_queueActive && _queueCount > 0)
153 _startNextQueuedAnimation();
154 else
155 {
156 _isLoopingNow = false;
157 _isRunning = false;
158 _currentFunc = nullptr;
159 _queueActive = false;
160 _currentQueueIndex = NO_QUEUE_INDEX;
161 _queueEntryDuration = 0;
162 _activeBlockEmissionStopAfter = 0;
163 }
164 }
165 }
166 return _isRunning;
167 }

305 SBK_BarMeterAnimations &enqueueReverseAnim()
306 {
307 if (_queueCount == 0)
308 return *this;
309
310 if (_queueCount >= QUEUE_CAPACITY)
311 {
312 _queueOverflow = true;
313 return *this;
314 }
315
316 const uint8_t previousIndex =
317 (_queueTail + QUEUE_CAPACITY - 1) % QUEUE_CAPACITY;
318 QueueEntry &entry = _queue[_queueTail];
319 entry = _queue[previousIndex];
320 entry.loop = _loop;
321 entry.animInitLogicIsInverted = !entry.animInitLogicIsInverted;
322 entry.animRenderLogicIsInverted = !entry.animRenderLogicIsInverted;
323 entry.resetRuntimeState = false;
324 entry.duration = 0;
325 entry.blockEmissionStopAfter = 0;
326
327 _queueTail = (_queueTail + 1) % QUEUE_CAPACITY;
328 _queueCount++;
329 return *this;
